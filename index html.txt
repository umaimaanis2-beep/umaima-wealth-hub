        let currentIdx = 0;
        let answers = new Array(questions.length).fill(null);

        function startAssessment() {
            document.getElementById('introScreen').style.display = 'none';
            document.getElementById('quizContainer').style.display = 'block';
            renderQuestion();
        }

        function renderQuestion() {
            const qObj = questions[currentIdx];
            const wrapper = document.getElementById('questionCardWrapper');
            
            let html = `<div class="question-title">${qObj.q}</div><div class="options-list">`;
            qObj.options.forEach((opt, idx) => {
                const checked = answers[currentIdx] === idx ? 'checked' : '';
                html += `
                    <label class="option-label">
                        <input type="radio" name="questionOption" value="${idx}" ${checked} onclick="selectOption(${idx})">
                        ${opt.text}
                    </label>
                `;
            });
            html += `</div>`;
            wrapper.innerHTML = html;

            // Update Progress
            const progress = ((currentIdx + 1) / questions.length) * 100;
            document.getElementById('progressFill').style.width = `${progress}%`;
            document.getElementById('questionCounter').innerText = `Question ${currentIdx + 1} of ${questions.length}`;

            // Button visibility
            document.getElementById('prevBtn').style.visibility = currentIdx === 0 ? 'hidden' : 'visible';
            document.getElementById('nextBtn').innerText = currentIdx === questions.length - 1 ? 'Finish' : 'Next';
        }

        function selectOption(idx) {
            answers[currentIdx] = idx;
        }

        function nextQuestion() {
            if (answers[currentIdx] === null) {
                alert("Please select an option before proceeding.");
                return;
            }

            if (currentIdx < questions.length - 1) {
                currentIdx++;
                renderQuestion();
            } else {
                showResults();
            }
        }

        function prevQuestion() {
            if (currentIdx > 0) {
                currentIdx--;
                renderQuestion();
            }
        }

        function showResults() {
            document.getElementById('quizContainer').style.display = 'none';
            document.getElementById('resultsScreen').style.display = 'block';

            const totalScore = answers.reduce((sum, optIdx, qIdx) => {
                return sum + questions[qIdx].options[optIdx].score;
            }, 0);

            document.getElementById('finalScore').innerText = `${totalScore} / 75`;

            let profile = "";
            if (totalScore >= 60) {
                profile = "Aggressive Growth Investor: You have a high risk appetite suited for long-term equity growth, high-beta assets, and expansion strategies.";
            } else if (totalScore >= 45) {
                profile = "Growth-Oriented Investor: You balance strong capital appreciation goals with manageable volatility buffers.";
            } else if (totalScore >= 30) {
                profile = "Balanced Investor: You seek a healthy equilibrium between steady income generation and moderate portfolio growth.";
            } else if (totalScore >= 18) {
                profile = "Conservative Investor: Capital preservation and minimizing drawdowns are your highest priorities.";
            } else {
                profile = "Risk-Averse Saver: You prioritize absolute safety, liquidity, and principal security above all else.";
            }

            document.getElementById('profileDescription').innerText = profile;
        }

        function resetAssessment() {
            currentIdx = 0;
            answers.fill(null);
            document.getElementById('resultsScreen').style.display = 'none';
            document.getElementById('introScreen').style.display = 'block';
